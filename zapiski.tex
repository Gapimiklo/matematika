\section{Kvadratna neenačba}

\begin{minipage}[t]{0.45\textwidth}
\vspace{0pt}
\begin{defbox}[equal height group=neenacba]{Definicija}
    \centering \large
    Neenačba ima oblike:
    $$ ax^2 + bx + c > 0 $$
    $$ ax^2 + bx + c < 0 $$
    $$ ax^2 + bx + c \leq 0 $$
    $$ ax^2 + bx + c \geq 0 $$ \\
    Kjer $a$ $\neq$ 0.\\ 
    Rešitev neenačbe je interval, unija intervalov ali prazna množica.
\end{defbox}
\end{minipage}
\hfill
\begin{minipage}[t]{0.45\textwidth}
\vspace{0pt}
\begin{zgledbox}[equal height group=neenacba]{Način reševanja}
    \centering
    $2x^2 < -6x + 8$\\
    $2x^2 +6x -8 < 0$\\
    $x^2 + 3x - 4 < 0$\\
    Tukaj rešimo kot kvadratno enačbo,\\ z Vietovim pravilom ali formulo za kvadratno enačbo.\\
    $x_1=-4$\\
    $x_2=1$\\
    Nato zapišemo interval. Če je znak je manjše(kot v tem primeru)\\ zapišemo interval, kdaj graf funkcije leži pod abscisno osjo.\\ Če je znak večje, zapišemo, kdaj graf leži nad abscisno osjo. Dobimo rešitev.\\[1mm]
    \large \boxed{$x \in (-4, 1)$}
\end{zgledbox}
\end{minipage}

\section{Kompleksna števila}
\begin{minipage}[t]{0.45\textwidth}
\vspace{0pt}
\begin{defbox}{Definicija}
    \centering
    Kompleksna števila, označena s simbolom $\mathbb{C}$, so vsa števila, ki jih lahko opišemo s predpisom:\\[8pt]
    $z=a+bi$\\[8pt]
    Kjer sta $a$ in $b$ realni števili in je $i^2$(imaginarna enota na kvadrat) enak $-1$\\[10pt]
    Vsako število lahko tudi razdelimo na realno in imaginarno komponento:\\
    $\text{Re}(z)=a$ in $\text{Im}(z)=b$
\end{defbox}
\begin{zgledbox}{Primer kompleksnega števila}
\centering
$z=4-8i$
\end{zgledbox}
\end{minipage}
\hfill
\begin{minipage}[t]{0.45\textwidth}
\vspace{0pt}

\end{minipage}

\subsection{Računanje s kompleksnimi števili}

\begin{minipage}[t]{0.45\textwidth}
\vspace{0pt}
\begin{defbox}[equal height group=racunanje]{Seštevanje in odštevanje}
\begin{align*}
z&=a+bi\\
w&=c+di\\
z+w=a+bi+&c+di=\boxed{a+c+(b+d)i}\\[8pt]
z-w=&a+bi-(c+di)=\\
a+bi-c-&di=\boxed{a-c+(b-d)i}
\end{align*}
\end{defbox}
\end{minipage}
\hfill
\begin{minipage}[t]{0.45\textwidth}
\vspace{0pt}
\begin{defbox}[equal height group=racunanje]{Množenje s skalarjem}
\begin{align*}
z=a+&bi \ \ \ k \in \mathbb{R}\\
z \cdot k = k(a+&bi)=ka+kbi
\end{align*}
\end{defbox}
\end{minipage}
